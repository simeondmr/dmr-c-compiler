Grammar:

<program> ::= { <function-declaration> }
<function-declaration> ::= "int" <identifier> "(" <function-params> ")" ( ";" | <block> )
<function-params> ::= "int" <identifier> { ","  "int" <identifier> } | "void"
<block> ::= "{" { <block-item> } "}"
<block-item> ::= <declaration> | <statement>
<for-init> ::= <var-declaration> | [ <expr> ] ";"
<statement> ::= "if" "(" <expr> ")" <statement> [ "else" <statement> ]
              | "while" "(" <expr>  ")" <statement>
              | "do" <statement> "while" "(" <expr> ")" ";"
              | "for" "(" <for-init> [ <expr> ] ";" [ <expr> ] ")" <stmt>
              | "switch" "(" <expr> ")"
              | "case" <int> ":"
              | "default" ":"
              | "break" ";"
              | "continue" ";"
              | "goto" <identifier> ";"
              | <identifier> ":" <statement>
              | "return" <exp> ";"
              | <expr> ";"
              | <block>
              | ";"
<declaration> ::= <var-declaration> | <function-declaration>
<var-declaration> ::= "int" <identifier> [ "=" <expr> ] ";"
<exp> ::= <factor>
        | <exp> binop <exp>
        | <exp> "?" <exp> ":" <exp>
<factor> ::= <int>
           | <unop> <exp>
           | <pre-post-op> <identifier>
           | <identifier> <pre-post-op>
           | "(" <exp> ")"
           | <identifier>
           | <identifier> "(" [ <func-args> ] ")"
<func-args> ::= <exp> { "," <exp> }
<unop> ::= "-" | "~" | "!"
<binop> ::= "+"
          | "-"
          | "*"
          | "/"
          | "%"
          | "&"
          | "|"
          | "^"
          | "~"
          | "<<"
          | ">>"
          | "=="
          | "!="
          | "<"
          | "<="
          | ">"
          | ">="
          | "&&"
          | "||"
          | "="
          | "+="
          | "-="
          | "*="
          | "/="
          | "|="
          | "&="
          | "^="
          | "<<="
          | ">>="
          | ","
<pre-post-op> ::= "++"
				| "--"
<identifier> ::= ? An identifier token ?
<int> ::= ? A constant token ?
